local Airflow
for _, url in ipairs(AIRFLOW_URLS) do
    local ok, result = pcall(function()
        local src = game:HttpGet(url)
        local fn, err = loadstring(src)
        if not fn then error("loadstring failed: " .. tostring(err)) end
        return fn()
    end)
    if ok and result then
        Airflow = result
        break
    else
        warn("[Zenith] Airflow load failed from " .. url .. " — " .. tostring(result))
    end
end

if not Airflow then
    error("[Zenith] Impossible de charger Airflow-UI. Vérifie ta connexion et que HttpGet est autorisé.")
end

-- Get the LocalPlayer's character root
local function GetRoot()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

-- Get the LocalPlayer's humanoid
local function GetHumanoid()
    local char = LocalPlayer.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

-- Check if a player is an ally/friend
local function IsFriend(player)
    return LocalPlayer:IsFriendsWith(player.UserId)
end

-- Check line of sight
local function HasLOS(origin, target)
    local ray = Ray.new(origin, (target - origin).Unit * 1000)
    local hit = Workspace:FindPartOnRayWithIgnoreList(ray, {LocalPlayer.Character})
    if hit and hit:IsDescendantOf(Workspace) then
        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character and hit:IsDescendantOf(player.Character) then
                return true
            end
        end
        return false
    end
    return true
end

-- Get closest enemy to Mouse / Screen Center
local function GetClosestEnemy(useFOV, fovRadius)
    local closest    = nil
    local closestDist = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if State.LegitFriendly and IsFriend(player) then continue end
        local char = player.Character
        if not char then continue end
        local part = char:FindFirstChild(State.LegitPart) or char:FindFirstChild("HumanoidRootPart")
        if not part then continue end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then continue end

        local screenPos, onScreen = Camera:WorldToScreenPoint(part.Position)
        if not onScreen then continue end
        local dist2d = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude

        if useFOV and dist2d > fovRadius then continue end

        if dist2d < closestDist then
            closestDist = dist2d
            closest = {player = player, part = part, dist2d = dist2d}
        end
    end
    return closest
end

-- ── Legit Aimbot ────────────────────────────────────────────
local function StartLegitAimbot()
    Disconnect("LegitAimbot")
    loops["LegitAimbot"] = RunService.RenderStepped:Connect(function()
        if not State.LegitAimbot then return end
        if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end

        local target = GetClosestEnemy(State.LegitFOV, State.LegitRadius)
        if not target then return end
        if State.LegitWallCheck then
            local root = GetRoot()
            if root and not HasLOS(root.Position, target.part.Position) then return end
        end

        local targetPos = target.part.Position
        -- Prediction offset
        local vel = target.part.AssemblyLinearVelocity
        targetPos = targetPos + vel * State.LegitPred

        local camCF   = Camera.CFrame
        local targetCF = CFrame.new(camCF.Position, targetPos)
        Camera.CFrame  = camCF:Lerp(targetCF, State.LegitSmooth)
    end)
end
StartLegitAimbot()

-- ── Flight ──────────────────────────────────────────────────
local flightBodyVelocity = nil
local flightBodyGyro     = nil

local function EnableFlight()
    local root = GetRoot()
    if not root then return end
    if flightBodyVelocity then flightBodyVelocity:Destroy() end
    if flightBodyGyro then flightBodyGyro:Destroy() end

    flightBodyVelocity          = Instance.new("BodyVelocity")
    flightBodyVelocity.Velocity = Vector3.zero
    flightBodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
    flightBodyVelocity.Parent   = root

    flightBodyGyro          = Instance.new("BodyGyro")
    flightBodyGyro.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
    flightBodyGyro.D         = 100
    flightBodyGyro.Parent    = root

    Disconnect("Flight")
    loops["Flight"] = RunService.RenderStepped:Connect(function()
        if not State.FlightEnable then
            if flightBodyVelocity then flightBodyVelocity:Destroy(); flightBodyVelocity = nil end
            if flightBodyGyro     then flightBodyGyro:Destroy(); flightBodyGyro = nil end
            Disconnect("Flight")
            return
        end
        local cf    = Camera.CFrame
        local vel   = Vector3.zero
        local speed = State.FlightSpeed
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then vel = vel + cf.LookVector * speed end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then vel = vel - cf.LookVector * speed end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then vel = vel - cf.RightVector * speed end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then vel = vel + cf.RightVector * speed end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vel = vel + Vector3.new(0, speed, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vel = vel - Vector3.new(0, speed, 0) end
        flightBodyVelocity.Velocity = vel
        flightBodyGyro.CFrame       = cf
    end)
end

local function DisableFlight()
    if flightBodyVelocity then flightBodyVelocity:Destroy(); flightBodyVelocity = nil end
    if flightBodyGyro     then flightBodyGyro:Destroy(); flightBodyGyro = nil end
    Disconnect("Flight")
end

local hitboxHighlights = {}
local function ApplyHitboxExpand()
    -- Clear old
    for _, h in pairs(hitboxHighlights) do h:Destroy() end
    hitboxHighlights = {}

    if not State.HitboxExpand then
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer then continue end
            local char = player.Character
            if not char then continue end
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then root.Size = Vector3.new(2, 2, 1) end
        end
        return
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        local char = player.Character
        if not char then continue end
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            root.Size = Vector3.new(State.HitboxSize, State.HitboxSize, State.HitboxSize)
            if State.HitboxVis then
                local sel = Instance.new("SelectionBox")
                sel.Adornee       = root
                sel.SurfaceColor3 = Color3.fromRGB(255, 0, 0)
                sel.LineThickness  = 0.05
                sel.Parent         = CoreGui
                hitboxHighlights[player.Name] = sel
            end
        end
    end
end

-- ── ESP ─────────────────────────────────────────────────────
local espHighlights = {}

local function UpdateESP()
    -- Clean stale entries
    for name, h in pairs(espHighlights) do
        if not Players:FindFirstChild(name) then
            h:Destroy()
            espHighlights[name] = nil
        end
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer and not State.ESPShowSelf then continue end
        local char = player.Character
        if not char then
            if espHighlights[player.Name] then
                espHighlights[player.Name]:Destroy()
                espHighlights[player.Name] = nil
            end
            continue
        end

        if State.ESPEnable then
            if not espHighlights[player.Name] then
                local h = Instance.new("Highlight")
                h.Adornee          = char
                h.FillTransparency = State.ESPBoxFilled and 0.5 or 1
                h.OutlineColor3    = Color3.fromRGB(255, 50, 50)
                h.FillColor3       = Color3.fromRGB(255, 50, 50)
                h.Parent           = CoreGui
                espHighlights[player.Name] = h
            else
                local h = espHighlights[player.Name]
                h.Adornee          = char
                h.FillTransparency = State.ESPBoxFilled and 0.5 or 1
            end
        else
            if espHighlights[player.Name] then
                espHighlights[player.Name]:Destroy()
                espHighlights[player.Name] = nil
            end
        end
    end
end

Disconnect("ESPUpdate")
loops["ESPUpdate"] = RunService.Heartbeat:Connect(UpdateESP)

            if not chamsHighlights[player.Name] then
                local h = Instance.new("Highlight")
                h.Adornee          = char
                h.FillColor3       = State.ChamsColor
                h.FillTransparency = 0.3
                h.OutlineTransparency = 1
                h.DepthMode        = Enum.HighlightDepthMode.AlwaysOnTop
                h.Parent           = CoreGui
                chamsHighlights[player.Name] = h
            else
                chamsHighlights[player.Name].FillColor3 = State.ChamsColor
            end
        else
            if chamsHighlights[player.Name] then
                chamsHighlights[player.Name]:Destroy()
                chamsHighlights[player.Name] = nil
            end
        end
    end
end

-- ── Crosshair ───────────────────────────────────────────────
local crosshairLines = {}
local hasDrawing = (Drawing ~= nil)
if hasDrawing then
    pcall(function()
        for i = 1, 4 do
            local line = Drawing.new("Line")
            line.Visible   = false
            line.Thickness = 2
            line.Color     = Color3.fromRGB(255, 255, 255)
            crosshairLines[i] = line
        end
    end)
end

local function UpdateCrosshair()
    if #crosshairLines == 0 then return end
    if not State.CrosshairEnable then
        for _, l in ipairs(crosshairLines) do l.Visible = false end
        return
    end
    local center = State.CrosshairFollow
        and Vector2.new(Mouse.X, Mouse.Y)
        or  Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    local gap  = State.CrosshairGap
    local len  = State.CrosshairLen
    local th   = State.CrosshairThick

    -- Up
    crosshairLines[1].From    = center - Vector2.new(0, gap)
    crosshairLines[1].To      = center - Vector2.new(0, gap + len)
    crosshairLines[1].Thickness = th
    crosshairLines[1].Visible   = true

    -- Down
    crosshairLines[2].From    = center + Vector2.new(0, gap)
    crosshairLines[2].To      = center + Vector2.new(0, gap + len)
    crosshairLines[2].Thickness = th
    crosshairLines[2].Visible   = true

    -- Left
    crosshairLines[3].From    = center - Vector2.new(gap, 0)
    crosshairLines[3].To      = center - Vector2.new(gap + len, 0)
    crosshairLines[3].Thickness = th
    crosshairLines[3].Visible   = true

    -- Right
    crosshairLines[4].From    = center + Vector2.new(gap, 0)
    crosshairLines[4].To      = center + Vector2.new(gap + len, 0)
    crosshairLines[4].Thickness = th
    crosshairLines[4].Visible   = true
end

Disconnect("Crosshair")
loops["Crosshair"] = RunService.RenderStepped:Connect(UpdateCrosshair)

-- ── Bloom & Post FX ─────────────────────────────────────────
local function ApplyPostFX()
    -- Bloom
    local bloom = Lighting:FindFirstChildOfClass("BlurEffect")
        or Lighting:FindFirstChild("Bloom")
        or Instance.new("BloomEffect", Lighting)
    if bloom:IsA("BloomEffect") then
        bloom.Enabled   = State.BloomEnable
        bloom.Intensity = State.BloomIntensity
        bloom.Size      = State.BloomSize
        bloom.Threshold = State.BloomThreshold
    end

    -- Color Correction
    local cc = Lighting:FindFirstChildOfClass("ColorCorrectionEffect")
        or Instance.new("ColorCorrectionEffect", Lighting)
    cc.Enabled    = State.ColorCorrEnable
    cc.Brightness = State.CCBrightness
    cc.Contrast   = State.CCContrast
    cc.Saturation = State.CCSaturation

    -- Sun Rays
    local sr = Lighting:FindFirstChildOfClass("SunRaysEffect")
        or Instance.new("SunRaysEffect", Lighting)
    sr.Enabled = State.SunRaysEnable

    -- No Shadows
    Lighting.GlobalShadows = not State.NoShadows

    -- No Skybox / Sky
    local sky = Lighting:FindFirstChildOfClass("Sky")
    if sky then sky.Parent = State.NoSkybox and game:GetService("ServerStorage") or Lighting end
end

    if State.WorldLighting then

    end
    if State.WorldTime then
        Lighting.ClockTime = State.WorldTimeVal
    end
    if State.WorldAtmo then
        local atmo = Lighting:FindFirstChildOfClass("Atmosphere")
            or Instance.new("Atmosphere", Lighting)
        atmo.Haze    = State.AtmoHaze
        atmo.Glare   = State.AtmoGlare
        atmo.Offset  = State.AtmoOffset
        atmo.Density = State.AtmoDensity
        atmo.Color   = State.AtmoColor1
        atmo.Decay   = State.AtmoColor2
    end
    if State.WorldAmbient then
        Lighting.Ambient    = State.AmbientColor1
        Lighting.OutdoorAmbient = State.AmbientColor2
    end
end

local function UpdateParticleAura()

    if not State.ParticleAura then return end
    local char = LocalPlayer.Character

    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local typeMap = {Sparkles = "Sparkles", Fire = "Fire", Smoke = "Smoke"}
    local pType   = typeMap[State.ParticleType] or "Sparkles"
    selfParticle  = Instance.new(pType)
    if pType == "Sparkles" then
        selfParticle.SparkleColor = State.ParticleColor
    elseif pType == "Fire" then
        selfParticle.Color = State.ParticleColor
    end
    selfParticle.Parent = root
end

            local elapsed = 0
            loops["ChatSpam"] = RunService.Heartbeat:Connect(function(dt)
                if not State.ChatSpam then Disconnect("ChatSpam") return end
                elapsed = elapsed + dt
                if elapsed >= State.ChatSpamDelay then
                    elapsed = 0
                    game:GetService("ReplicatedStorage")
                        :FindFirstChild("DefaultChatSystemChatEvents")
                        and game:GetService("ReplicatedStorage")
                            .DefaultChatSystemChatEvents
                            .SayMessageRequest
                            :FireServer(State.ChatSpamMsg, "All")
                end
            end)
        else
            Disconnect("ChatSpam")
        end
    end,
})

local UnloadSection = SettingsTab:CreateSection({ Title = "── UNLOAD ──" })

SettingsTab:CreateButton({
    Section  = UnloadSection,
    Title    = "Unload Zenith",
    Callback = function()
        -- Clean up all loops and drawn objects
        for name, conn in pairs(loops) do
            conn:Disconnect()
        end
        fovCircle:Remove()
        for _, line in ipairs(crosshairLines) do line:Remove() end
        for _, h in pairs(espHighlights) do h:Destroy() end
        for _, h in pairs(chamsHighlights) do h:Destroy() end
        for _, h in pairs(hitboxHighlights) do h:Destroy() end
        if selfParticle then selfParticle:Destroy() end
        if flightBodyVelocity then flightBodyVelocity:Destroy() end
        if flightBodyGyro then flightBodyGyro:Destroy() end

        -- Restore defaults
        local hum = GetHumanoid()
        if hum then
            hum.WalkSpeed  = 16
            hum.JumpPower  = 50
            hum.MaxHealth  = 100
        end
        Workspace.Gravity            = 196.2
        Lighting.GlobalShadows       = true
        Camera.FieldOfView           = 70

        -- Destroy UI
        Window:Destroy()
    end,
})
