-- ── Noclip ────────────────────────────────────────────────────────────────
TabPlayer:CreateSection("Physique")

local noclipActive = false
local noclipConnection

TabPlayer:CreateToggle({
    Name     = "Noclip",
    Desc     = "Passe à travers les murs",
    Flag     = "Noclip",
    Callback = function(Value)
        noclipActive = Value
        if Value then
            noclipConnection = RunService.Stepped:Connect(function()
                if Character then
                    for _, part in ipairs(Character:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.CanCollide = false
                        end
                    end
                end
            end)
        else
            if noclipConnection then
                noclipConnection:Disconnect()
                noclipConnection = nil
            end
            -- Réactive la collision
            for _, part in ipairs(Character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = true
                end
            end
        end
    end,
})

local infiniteJumpActive = false
local ijConn

TabPlayer:CreateToggle({
    Name     = "Infinite Jump",
    Desc     = "Sauter même en l'air",
    Flag     = "InfiniteJump",
    Callback = function(Value)
        infiniteJumpActive = Value
        if Value then
            ijConn = UserInputService.JumpRequest:Connect(function()
                if infiniteJumpActive and Humanoid then
                    Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                end
            end)
        else
            if ijConn then
                ijConn:Disconnect()
                ijConn = nil
            end
        end
    end,
})

        if RootPart then
            if spawnLocation then

            else
                RootPart.CFrame = CFrame.new(0, 10, 0)
            end

                Title    = "Téléporté",
                Content  = "Arrivée au spawn.",
                Icon     = "map-pin",

local tpPlayerInput = TabPlayer:CreateInput({
    Name            = "Téléporter vers joueur",
    Desc            = "Nom du joueur cible",
    Icon            = "user",
    PlaceholderText = "ex : Pookie123",
    Flag            = "TpTargetPlayer",
    Callback        = function(Text, Enter)
        if not Enter then return end
        local target = Players:FindFirstChild(Text)
            or (function()
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Name:lower():find(Text:lower()) then
                        return p
                    end
                end
            end)()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            RootPart.CFrame = target.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
            Airflow:Notify({
                Title    = "Téléporté",
                Content  = "Arrivée près de " .. target.Name,
                Icon     = "user",
                Type     = "Success",
                Duration = 3,
            })
        else
            Airflow:Notify({
                Title    = "Introuvable",
                Content  = "Aucun joueur correspondant à \"" .. Text .. "\".",
                Icon     = "alert-triangle",
                Type     = "Warning",
                Duration = 3,
            })
        end
    end,
})

local TabVisual = Window:CreateTab({
    Name = "Visual",
    Desc = "Effets visuels & caméra",
    Icon = "eye",
})

TabVisual:CreateSection("Joueurs")

local espActive = false
local espHighlights = {}

local function removeESP()
    for _, h in pairs(espHighlights) do
        if h and h.Parent then h:Destroy() end
    end
    espHighlights = {}
end

local function buildESP()
    removeESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local h = Instance.new("Highlight")
            h.FillColor   = Color3.fromRGB(140, 180, 255)
            h.OutlineColor = Color3.fromRGB(255, 255, 255)
            h.FillTransparency    = 0.5
            h.OutlineTransparency = 0
            h.Adornee = player.Character
            h.Parent  = player.Character
            espHighlights[player.Name] = h
        end
    end
end

TabVisual:CreateToggle({
    Name     = "ESP Joueurs",
    Desc     = "Highlight tous les joueurs",
    Flag     = "ESP",
    Callback = function(Value)
        espActive = Value
        if Value then
            buildESP()
        else
            removeESP()
        end
    end,
})

-- Refresh ESP quand un joueur apparaît
Players.PlayerAdded:Connect(function(player)
    if not espActive then return end
    player.CharacterAdded:Connect(function(char)
        task.wait(1)
        if espActive then buildESP() end
    end)
end)

Players.PlayerRemoving:Connect(function(player)
    if espHighlights[player.Name] then
        espHighlights[player.Name]:Destroy()
        espHighlights[player.Name] = nil
    end
end)

-- Couleur de l'ESP
TabVisual:CreateColorPicker({
    Name     = "Couleur ESP",
    Desc     = "Teinte du highlight",
    Color    = Color3.fromRGB(140, 180, 255),
    Flag     = "EspColor",
    Callback = function(Color)
        for _, h in pairs(espHighlights) do
            h.FillColor = Color
        end
    end,
})

TabVisual:CreateSection("Environnement")

TabVisual:CreateSlider({
    Name         = "Field of View",

            Lighting.OutdoorAmbient   = Color3.fromRGB(120, 120, 120)
            Lighting.Brightness       = brightnessBefore

    end,
